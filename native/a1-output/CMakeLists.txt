cmake_minimum_required(VERSION 3.24)
project(A1NdiSender VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
add_executable(a1-ndi-sender src/main.cpp src/desktop.cpp src/voicemeeter.cpp)
target_include_directories(a1-ndi-sender PRIVATE src)
target_include_directories(a1-ndi-sender SYSTEM PRIVATE third_party)
target_compile_definitions(a1-ndi-sender PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(a1-ndi-sender PRIVATE /W4 /wd4324 /permissive- /utf-8 /EHsc)
target_link_libraries(a1-ndi-sender PRIVATE d3d11 dxgi d3dcompiler avrt winmm shell32 user32 ole32 psapi)
enable_testing()
add_executable(timing-tests tests/timing_tests.cpp)
target_include_directories(timing-tests PRIVATE src)
target_compile_definitions(timing-tests PRIVATE NOMINMAX)
add_test(NAME timing-and-queues COMMAND timing-tests)
add_executable(config-tests tests/config_tests.cpp)
target_include_directories(config-tests PRIVATE src)
target_compile_definitions(config-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_test(NAME output-settings COMMAND config-tests)

option(A1_BUILD_OBS_PLUGIN "Build the local OBS source (OBS 32.0.1 headers)" OFF)
set(A1_OBS_SOURCE_DIR "" CACHE PATH "Extracted official OBS 32.0.1 source directory")
if(A1_BUILD_OBS_PLUGIN)
  if(NOT EXISTS "${A1_OBS_SOURCE_DIR}/libobs/obs-module.h")
    message(FATAL_ERROR "Use Build-OBSPlugin.ps1 to prepare official OBS headers")
  endif()
  set(OBS_IMPORT_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/obs-import.lib")
  add_custom_command(OUTPUT "${OBS_IMPORT_LIBRARY}"
    COMMAND "${CMAKE_AR}" /NOLOGO /MACHINE:X64
      "/DEF:${CMAKE_CURRENT_SOURCE_DIR}/obs-plugin/obs.def" "/OUT:${OBS_IMPORT_LIBRARY}"
    DEPENDS obs-plugin/obs.def VERBATIM)
  set(OBS_FRONTEND_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/frontend-import.lib")
  add_custom_command(OUTPUT "${OBS_FRONTEND_LIBRARY}"
    COMMAND "${CMAKE_AR}" /NOLOGO /MACHINE:X64
      "/DEF:${CMAKE_CURRENT_SOURCE_DIR}/obs-plugin/frontend.def" "/OUT:${OBS_FRONTEND_LIBRARY}"
    DEPENDS obs-plugin/frontend.def VERBATIM)
  add_library(a1-local-source MODULE obs-plugin/a1_local_source.cpp obs-plugin/deck_control.cpp src/desktop.cpp src/voicemeeter.cpp "${OBS_IMPORT_LIBRARY}" "${OBS_FRONTEND_LIBRARY}")
  target_include_directories(a1-local-source PRIVATE src obs-plugin)
  target_include_directories(a1-local-source SYSTEM PRIVATE third_party "${A1_OBS_SOURCE_DIR}/libobs" "${A1_OBS_SOURCE_DIR}/frontend/api")
  target_compile_definitions(a1-local-source PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(a1-local-source PRIVATE /W4 /wd4324 /permissive- /utf-8 /EHsc)
  target_link_libraries(a1-local-source PRIVATE "${OBS_IMPORT_LIBRARY}" "${OBS_FRONTEND_LIBRARY}" d3d11 dxgi d3dcompiler avrt winmm shell32 user32 ole32)
endif()
